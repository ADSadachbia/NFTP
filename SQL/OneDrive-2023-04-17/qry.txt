CREATE TABLE estudiantes (
  ID INT IDENTITY(1,1) PRIMARY KEY,
  ID_Curso INT,
  codigo_estudiante Varchar(35),
  nombre VARCHAR(100),
  apellido VARCHAR(100),
  edad INT,
  telefono VARCHAR(100),
  cedula VARCHAR(20),
  region VARCHAR(20)
  FOREIGN KEY (ID_Curso) REFERENCES Cursos(ID)
  );
