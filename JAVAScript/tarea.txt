createElement 
// Create a new <div> element
var div = document.createElement("div");

// Set attributes and content for the element
div.id = "myDiv";
div.className = "myClass";
div.innerHTML = "This is a dynamically created <div> element.";

// Append the element to an existing element in the document
var container = document.getElementById("container");
container.appendChild(div);

getElementbyName
<!DOCTYPE html>
<html>
<head>
  <title>getElementsByName Example</title>
</head>
<body>
  <input type="text" name="username">
  <input type="text" name="username">
  <input type="text" name="username">

  <script>
    // Acceder a todos los elementos con el nombre "username"
    var elements = document.getElementsByName("username");

    // Iterar sobre la colección de elementos y realizar modificaciones
    for (var i = 0; i < elements.length; i++) {
      var input = elements[i];
      input.value = "John Doe";
      input.style.backgroundColor = "lightblue";
    }
  </script>
</body>
</html>

getelementbytagname

<!DOCTYPE html>
<html>
<head>
  <title>getElementsByTagName Example</title>
</head>
<body>
  <h1>Heading 1</h1>
  <p>Paragraph 1</p>
  <h2>Heading 2</h2>
  <p>Paragraph 2</p>
  <h3>Heading 3</h3>
  <p>Paragraph 3</p>

  <script>
    // Acceder a todos los elementos <h2>
    var headings = document.getElementsByTagName("h2");

    // Iterar sobre la colección de elementos y realizar modificaciones
    for (var i = 0; i < headings.length; i++) {
      var heading = headings[i];
      heading.style.color = "blue";
    }
  </script>
</body>
</html>
